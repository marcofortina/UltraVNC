project(winvnc_portable_tests LANGUAGES CXX C)

find_package(ZLIB REQUIRED)
find_package(JPEG REQUIRED)

function(uvnc_add_winvnc_portable_test name source)
    add_executable(${name}
        "${CMAKE_SOURCE_DIR}/tests/${source}"
    )

    target_link_libraries(${name} PRIVATE libwinvncportablecore ZLIB::ZLIB JPEG::JPEG)

    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name}
        PROPERTIES
            LABELS "native-linux;winvnc-portable"
            TIMEOUT 10
    )
endfunction()

function(uvnc_add_viewer_portable_test name source)
    add_executable(${name}
        "${CMAKE_SOURCE_DIR}/tests/${source}"
        "${CMAKE_SOURCE_DIR}/../vncviewer/vncPortableViewerConfig.cpp"
        "${CMAKE_SOURCE_DIR}/../vncviewer/vncPortableViewerCli.cpp"
        "${CMAKE_SOURCE_DIR}/../vncviewer/vncPortableViewerSession.cpp"
        "${CMAKE_SOURCE_DIR}/../vncviewer/vncPortableVncAuth.cpp"
        "${CMAKE_SOURCE_DIR}/../rfb/vncauth.c"
        "${CMAKE_SOURCE_DIR}/../common/d3des.c"
    )

    target_include_directories(${name} PRIVATE
        "${CMAKE_SOURCE_DIR}/../vncviewer"
        "${CMAKE_SOURCE_DIR}/../winvnc/winvnc"
        "${CMAKE_SOURCE_DIR}/.."
        "${CMAKE_SOURCE_DIR}/../common"
    )
    target_link_libraries(${name} PRIVATE libwinvncportablecore ZLIB::ZLIB JPEG::JPEG)

    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name}
        PROPERTIES
            LABELS "native-linux;vncviewer-portable"
            TIMEOUT 10
    )
endfunction()
